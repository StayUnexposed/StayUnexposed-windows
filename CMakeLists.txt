cmake_minimum_required(VERSION 3.24)

# The app version: the VERSION file is the single source (major.minor.patch).
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" SU_VERSION LIMIT_COUNT 1)
string(STRIP "${SU_VERSION}" SU_VERSION)
project(StayUnexposed VERSION ${SU_VERSION} LANGUAGES C CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
configure_file(src/Version.h.in "${CMAKE_BINARY_DIR}/generated/Version.h" @ONLY)
# The exe's resources are compiled through this generated wrapper, which names the version: Visual Studio
# doesn't notice when the generated Version.h changes, but it does recompile a changed .rc, so the file
# properties never keep an old version number.
# The hashes make edits to res/app.rc or the icon change the wrapper too.
file(SHA1 "${CMAKE_CURRENT_SOURCE_DIR}/res/app.rc" SU_RC_HASH)
file(SHA1 "${CMAKE_CURRENT_SOURCE_DIR}/res/app.ico" SU_ICO_HASH)
file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/generated/app_version.rc" CONTENT
  "// Generated by CMake (version ${SU_VERSION}, app.rc ${SU_RC_HASH}, app.ico ${SU_ICO_HASH}); edit res/app.rc instead.\n#include \"${CMAKE_CURRENT_SOURCE_DIR}/res/app.rc\"\n")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/res/app.rc" "${CMAKE_CURRENT_SOURCE_DIR}/res/app.ico")
message(STATUS "Stay Unexposed version ${PROJECT_VERSION}")

if(NOT WIN32)
  message(FATAL_ERROR "This application targets Windows only.")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")  # static CRT -> single exe

# ---------------------------------------------------------------------------
# Release size/speed: smaller single exe (applies to every target defined below,
# including the fetched imgui/sqlite). Debug builds are left untouched.
#   /O1 /Os  - optimise for size (this app is not CPU-bound)
#   /Gy /Gw  - one COMDAT per function and per global, so the linker can drop unused ones
#   LTCG     - whole-program optimisation and cross-module inlining (/GL + /LTCG)
#   /OPT:REF - remove unreferenced functions/data;  /OPT:ICF - merge identical ones
# ---------------------------------------------------------------------------
if(MSVC)
  add_compile_options($<$<CONFIG:Release>:/O1> $<$<CONFIG:Release>:/Os>
                      $<$<CONFIG:Release>:/Gy> $<$<CONFIG:Release>:/Gw>)
  add_link_options($<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
endif()

# ---------------------------------------------------------------------------
# Third-party dependencies (downloaded at configure time)
# ---------------------------------------------------------------------------
include(FetchContent)
set(CMAKE_POLICY_DEFAULT_CMP0135 NEW)  # extracted files get the extraction timestamp

FetchContent_Declare(imgui
  URL https://github.com/ocornut/imgui/archive/refs/tags/v1.91.5.tar.gz)
FetchContent_Declare(json
  URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
FetchContent_Declare(sqlite
  URL https://www.sqlite.org/2024/sqlite-amalgamation-3460100.zip)
# imgui and the sqlite amalgamation have no CMakeLists.txt: MakeAvailable just downloads them.
FetchContent_MakeAvailable(json imgui sqlite)

add_library(imgui STATIC
  ${imgui_SOURCE_DIR}/imgui.cpp
  ${imgui_SOURCE_DIR}/imgui_draw.cpp
  ${imgui_SOURCE_DIR}/imgui_tables.cpp
  ${imgui_SOURCE_DIR}/imgui_widgets.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_dx11.cpp
  ${imgui_SOURCE_DIR}/misc/cpp/imgui_stdlib.cpp)
target_include_directories(imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends ${imgui_SOURCE_DIR}/misc/cpp)
target_compile_definitions(imgui PUBLIC IMGUI_DISABLE_OBSOLETE_FUNCTIONS)

add_library(sqlite3 STATIC ${sqlite_SOURCE_DIR}/sqlite3.c)
target_include_directories(sqlite3 PUBLIC ${sqlite_SOURCE_DIR})
# Compile out the SQLite features the app never uses (prepared statements + exec on UTF-8 only:
# no UTF-16, FTS, JSON, blobs, custom functions/collations or extensions). Smaller code, same behaviour.
target_compile_definitions(sqlite3 PRIVATE
  SQLITE_THREADSAFE=1 SQLITE_DQS=0
  SQLITE_OMIT_LOAD_EXTENSION SQLITE_OMIT_UTF16 SQLITE_OMIT_DEPRECATED
  SQLITE_OMIT_DECLTYPE SQLITE_OMIT_PROGRESS_CALLBACK SQLITE_OMIT_SHARED_CACHE
  SQLITE_OMIT_GET_TABLE SQLITE_OMIT_COMPLETE SQLITE_OMIT_TCL_VARIABLE SQLITE_OMIT_TRACE
  SQLITE_DEFAULT_MEMSTATUS=0 SQLITE_USE_ALLOCA SQLITE_MAX_EXPR_DEPTH=0 SQLITE_UNTESTABLE)

# ---------------------------------------------------------------------------
# Core library: API, models, storage, watch logic (no UI) -- also used by tests
# ---------------------------------------------------------------------------
add_library(su_core STATIC
  src/core/Util.cpp
  src/core/Database.cpp
  src/core/Autostart.cpp
  src/core/HardwareId.cpp
  src/core/DownloadStamp.cpp
  src/core/Signature.cpp
  src/api/HttpClient.cpp
  src/api/ApiClient.cpp
  src/api/MockData.cpp
  src/api/Models.cpp
  src/services/WatchService.cpp
  src/services/Exporter.cpp
  src/services/ReferralClient.cpp
  src/services/Updater.cpp
  src/services/AgentRunner.cpp)
target_include_directories(su_core PUBLIC src "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(su_core PUBLIC nlohmann_json::nlohmann_json sqlite3 winhttp shell32 ole32 iphlpapi bcrypt wintrust crypt32)
target_compile_definitions(su_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN SU_MOCK)

# ---------------------------------------------------------------------------
# Application
# ---------------------------------------------------------------------------
add_executable(StayUnexposed WIN32
  src/main.cpp
  src/ui/App.cpp
  src/ui/SearchView.cpp
  src/ui/WatchlistView.cpp
  src/ui/ResultsTable.cpp
  src/ui/IndexGraph.cpp
  src/ui/AdvancedForm.cpp
  src/ui/ReferralView.cpp
  src/ui/Tray.cpp
  src/ui/Style.cpp
  src/ui/Widgets.cpp
  "${CMAKE_BINARY_DIR}/generated/app_version.rc")
target_include_directories(StayUnexposed PRIVATE res)  # app.rc's resource.h and app.ico
target_link_libraries(StayUnexposed PRIVATE su_core imgui d3d11 dxgi d3dcompiler dwmapi shell32)

if(MSVC)
  foreach(t su_core StayUnexposed)
    target_compile_options(${t} PRIVATE /W4 /utf-8 /permissive-)
  endforeach()
  target_compile_options(imgui PRIVATE /utf-8)
endif()

# ---------------------------------------------------------------------------
# Tests
# ---------------------------------------------------------------------------
enable_testing()
add_executable(su_tests tests/test_core.cpp)
target_link_libraries(su_tests PRIVATE su_core)
if(MSVC)
  target_compile_options(su_tests PRIVATE /W4 /utf-8 /permissive-)
endif()
add_test(NAME su_tests COMMAND su_tests)
